
# to enter credentials locally
aws configure


AWS Access Key ID [None]: YOUR_ACCESS_KEY_ID
AWS Secret Access Key [None]: YOUR_SECRET_ACCESS_KEY
Default region name [None]: eu-central-1      # or us-west-2, etc.
Default output format [None]: json            # or text, table



Default Output Format
Choose how AWS CLI returns results:

json → Recommended for automation
text → For plain output
table → Pretty-printed tables


This will create or update two files:
~/.aws/credentials — stores the secret credentials.
~/.aws/config — stores the region/output format.

# to change configurations
aws configure set region eu-west-1


# to see configurations
aws configure list


# we can set up more than 1 profile if necessary
# for that we need to save additional configurations into both credentials & config files manually

[user1]
aws_access_key_id=AKIAI44QH8DHBEXAMPLE
aws_secret_access_key=je7MtGbClwBF/2Zp9Utk/h3yCo8nvbEXAMPLEKEY

[user2]
aws_access_key_id=AKIAI44QH8DHBEXAMPLE
aws_secret_access_key=je7MtGbClwBF/2Zp9Utk/h3yCo8nvbEXAMPLEKEY


[profile user1]
region=us-east-1
output=text


[profile user2]
region=us-east-1
output=text



# To use the user1 profile in the AWS CLI:
aws s3 ls --profile user1


# Or, to make it the default temporarily in your terminal:
export AWS_PROFILE=user1


# View individual values:
aws configure get region --profile user1


# Set a config value programmatically:
aws configure set region eu-west-1 --profile user1
