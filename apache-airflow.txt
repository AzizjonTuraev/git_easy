# to fetch apache airflow docker file using docker 
curl 'https://airflow.apache.org/docs/apache-airflow/2.10.5/docker-compose.yaml' -o 'docker-compose.yaml'


mkdir logs
mkdir dags
mkdir plugins

docker compose up airflow-init
docker compose up
docker compose down
